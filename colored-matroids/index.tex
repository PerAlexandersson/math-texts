\section[chapters]{Chapters}

\begin{enumerate}
\item \href{intro.htm}{\textbf{Introduction}}. Words, ordered set partitions,
and symmetric cycle exchange.

\item \href{constructions.htm}{\textbf{Constructions and examples}}. Allowed
colors, fixed-indegree orientations, interval lattice paths, and
representable lifts.

\item \href{lattice-paths.htm}{\textbf{Colored lattice-path matroids}}.
Prefix-count bounds, path polytopes, and regions in the quota box.

\item \href{operations.htm}{\textbf{Operations and minors}}. Loops and
coloops, restrictions, color-indexed minors that recover deletion and
contraction, and direct sums.

\item \href{tutte.htm}{\textbf{Distance and Tutte polynomials}}. A distance
enumerator, its tropical minor recurrence, and the classical Tutte polynomial
in two colors.

\item \href{log-concavity.htm}{\textbf{Ultra-log-concavity}}. Rigid
width-three posets and three-color matroid encodings.

\item \href{rank.htm}{\textbf{Rank}}. Ordinary color projections, joint cell
counts, rank data, and the boundary of current rank descriptions.

\item \href{polytope.htm}{\textbf{Assignment polytopes}}. Assignment-vector
geometry, edge directions, inequality descriptions, and the principal open
questions.
\end{enumerate}
